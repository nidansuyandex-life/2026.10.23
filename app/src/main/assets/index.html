' ========================================================== ' 主程序：合同导出、转图片、多图合并PDF并高亮、清空模板 ' ========================================================== Sub 合同导出与保存20266() Dim ws As Worksheet Dim fileName As String, baseName As String Dim dateStr As String, timeStr As String, fullDateTimeStr As String Dim serialNum As Long Dim fsoCheck As Object, fso As Object, fileItem As Object Dim jpgFolderPath As String, jpgFilePath As String ' 设置当前工作表 Set ws = ActiveSheet ' 获取当前日期和时间（用于备用文件名） dateStr = Format(Date, "yyyymmdd") timeStr = Format(Time, "hhmmss") fullDateTimeStr = dateStr & "_" & timeStr ' --- 构建文件名和编号 --- baseName = ws.Range("B4").Value & ws.Range("N3").Value baseName = CleanFileName(baseName) fileName = "G:\临时文件夹\合同夹\" & baseName & ".pdf" ' --- 保存原文件 --- ThisWorkbook.Save ' --- 导出PDF --- On Error Resume Next ws.ExportAsFixedFormat Type:=xlTypePDF, fileName:=fileName, Quality:=xlQualityStandard If Err.Number <> 0 Then baseName = ws.Range("B4").Value & "_" & fullDateTimeStr & "_" & serialNum fileName = "G:\临时文件夹\合同夹\" & baseName & ".pdf" ws.ExportAsFixedFormat Type:=xlTypePDF, fileName:=fileName, Quality:=xlQualityStandard End If On Error GoTo 0 ' 检查PDF是否生成成功 Set fsoCheck = CreateObject("Scripting.FileSystemObject") If Not fsoCheck.FileExists(fileName) Then MsgBox "PDF导出失败，无法继续后续操作。", vbExclamation GoTo SkipConversion End If ' ========== 检查并创建“图片合同”文件夹 ========== jpgFolderPath = "G:\临时文件夹\合同夹\图片合同" If Not fsoCheck.FolderExists(jpgFolderPath) Then fsoCheck.CreateFolder jpgFolderPath End If ' ========== 记录转换前的旧图片 ========== Dim oldImages As Object Set oldImages = CreateObject("Scripting.Dictionary") Set fso = CreateObject("Scripting.FileSystemObject") For Each fileItem In fso.GetFolder(jpgFolderPath).Files oldImages(fileItem.Name) = True Next fileItem ' ========== 将PDF转为JPG图片 ========== jpgFilePath = ConvertPDFToJPG(fileName, jpgFolderPath) ' ========== 收集刚刚生成的新图片 ========== Dim newImages As Collection Set newImages = New Collection For Each fileItem In fso.GetFolder(jpgFolderPath).Files If Not oldImages.Exists(fileItem.Name) Then Dim ext As String ext = LCase(fso.GetExtensionName(fileItem.Name)) If ext = "jpg" Or ext = "jpeg" Or ext = "png" Or ext = "bmp" Then newImages.Add fileItem.Path End If End If Next fileItem ' ========== 将新图片合并为完整 PDF 并高亮 ========== If newImages.Count > 0 Then Call 多张图片合并PDF并高亮(newImages, "G:\临时文件夹\合同夹") End If SkipConversion: Set fsoCheck = Nothing Set fso = Nothing ' ========== 清空当前工作表并填入默认值 ========== With ws .Range("B4:J4").ClearContents .Range("N3:O3").ClearContents .Range("W7:Z14").ClearContents .Range("M7:O15").ClearContents .Range("M40:O63").ClearContents .Range("T40:X62").ClearContents .Range("K7").Formula = "=G7*I7" .Range("B7").Formula = "=IF(W7="""","""",AA7)" .Range("G7").Formula = "=IF(W7="""","""",Y7)" .Range("I7").Formula = "=IF(W7="""","""",Z7)" .Range("C17:O17").Value = "" .Range("A16:O16").Value = "二、质量和技术要求：供方承诺所提供产品均按照国家行业标准以及需方备注及要求执行：" .Range("C19:O19").Value = "简易布袋装，包装不回收" .Range("D21:O21").Value = "款到发货" .Range("E18").Value = "物流运输，需方指定地点并需方负责运输费用" .Range("B4").Select End With ThisWorkbook.Save End Sub ' ========================================================== ' 新增子程序：多张图片合并为一个 PDF 并高亮 ' ========================================================== Sub 多张图片合并PDF并高亮(imgPaths As Collection, destFolder As String) Dim fso As Object Dim baseName As String, finalPdfPath As String Dim counter As Integer, i As Integer, j As Integer Dim tempStr As String Dim AcroExchApp As Object Dim tempPdfPaths As Collection Dim tempPdfPath As String Dim arr() As String Set fso = CreateObject("Scripting.FileSystemObject") Set tempPdfPaths = New Collection ' 将 Collection 转为数组并排序（确保多页顺序正确） ReDim arr(1 To imgPaths.Count) For i = 1 To imgPaths.Count arr(i) = imgPaths(i) Next i ' 简单的冒泡排序 For i = 1 To UBound(arr) - 1 For j = i + 1 To UBound(arr) If arr(i) > arr(j) Then tempStr = arr(i) arr(i) = arr(j) arr(j) = tempStr End If Next j Next i ' 构建最终 PDF 路径（处理重名） baseName = fso.GetBaseName(arr(1)) ' 如果图片名带有 _1, _2 后缀，去掉它作为基础名 If InStr(baseName, "_") > 0 Then Dim lastPart As String lastPart = Mid(baseName, InStrRev(baseName, "_") + 1) If IsNumeric(lastPart) Then baseName = Left(baseName, InStrRev(baseName, "_") - 1) End If End If finalPdfPath = destFolder & "\" & baseName & ".pdf" If fso.FileExists(finalPdfPath) Then counter = 1 Do finalPdfPath = destFolder & "\" & baseName & " (" & counter & ").pdf" counter = counter + 1 Loop While fso.FileExists(finalPdfPath) End If ' 启动 Acrobat 进程 On Error Resume Next Set AcroExchApp = CreateObject("AcroExch.App") If AcroExchApp Is Nothing Then MsgBox "无法启动 Adobe Acrobat，请确保已正确安装！", vbCritical Exit Sub End If On Error GoTo 0 ' 1. 把每张图片转成临时单页 PDF For i = 1 To UBound(arr) tempPdfPath = destFolder & "\_temp_" & i & "_" & Format(Now, "hhmmss") & ".pdf" If ConvertSingleImgToPDF(arr(i), tempPdfPath, AcroExchApp) Then tempPdfPaths.Add tempPdfPath End If Next i ' 2. 合并这些临时 PDF If tempPdfPaths.Count > 0 Then Dim mainPDDoc As Object, subPDDoc As Object Set mainPDDoc = CreateObject("AcroExch.PDDoc") If mainPDDoc.Open(tempPdfPaths(1)) Then For i = 2 To tempPdfPaths.Count Set subPDDoc = CreateObject("AcroExch.PDDoc") If subPDDoc.Open(tempPdfPaths(i)) Then ' 插入到现有文档的最后一页之后 mainPDDoc.InsertPages mainPDDoc.GetNumPages - 1, subPDDoc, 0, subPDDoc.GetNumPages, False subPDDoc.Close End If Set subPDDoc = Nothing Next i ' 保存合并后的 PDF mainPDDoc.Save 1, finalPdfPath mainPDDoc.Close End If Set mainPDDoc = Nothing End If ' 3. 清理临时文件 For i = 1 To tempPdfPaths.Count If fso.FileExists(tempPdfPaths(i)) Then fso.DeleteFile tempPdfPaths(i), True End If Next i ' 4. 退出 Acrobat 并高亮 AcroExchApp.Exit Set AcroExchApp = Nothing Set fso = Nothing If fso Is Nothing Then Set fso = CreateObject("Scripting.FileSystemObject") If fso.FileExists(finalPdfPath) Then Shell "explorer.exe /select," & finalPdfPath, vbNormalFocus End If Set fso = Nothing End Sub ' ========================================================== ' 辅助函数：单张图片转单页 PDF ' ========================================================== Function ConvertSingleImgToPDF(imgPath As String, pdfPath As String, ByVal AcroExchApp As Object) As Boolean Dim AcroExchAVDoc As Object Set AcroExchAVDoc = CreateObject("AcroExch.AVDoc") ConvertSingleImgToPDF = False If AcroExchAVDoc.Open(imgPath, "") Then AcroExchAVDoc.SaveAs pdfPath, "com.adobe.acrobat.pdf" AcroExchAVDoc.Close True ConvertSingleImgToPDF = True End If Set AcroExchAVDoc = Nothing End Function ' ========================================================== ' 辅助函数：PDF 转 JPG 图片（返回生成的 JPG 完整路径） ' ========================================================== Function ConvertPDFToJPG(pdfPath As String, jpgFolder As String) As String Dim FilePathJPG As String Dim pdfDoc As Object Dim AcroExchApp As Object Dim AcroExchAVDoc As Object Dim jso As Object Dim fso As Object Dim baseName As String Set fso = CreateObject("Scripting.FileSystemObject") baseName = fso.GetBaseName(pdfPath) If Right(jpgFolder, 1) = "\" Then FilePathJPG = jpgFolder & baseName & ".jpg" Else FilePathJPG = jpgFolder & "\" & baseName & ".jpg" End If On Error GoTo ErrorHandler Set AcroExchApp = CreateObject("AcroExch.App") Set AcroExchAVDoc = CreateObject("AcroExch.AVDoc") If AcroExchAVDoc.Open(pdfPath, "") <> True Then MsgBox "转换图片失败：无法打开PDF文件。" & vbCrLf & pdfPath, vbExclamation GoTo CleanUp End If Set pdfDoc = AcroExchAVDoc.GetPDDoc() Set jso = pdfDoc.GetJSObject ' 注意：Acrobat 的多页 PDF 转 JPG，多页时可能会自动生成多个文件（如 _1.jpg, _2.jpg） jso.SaveAs FilePathJPG, "com.adobe.acrobat.jpeg" CleanUp: If Not AcroExchAVDoc Is Nothing Then AcroExchAVDoc.Close (True) If Not AcroExchApp Is Nothing Then AcroExchApp.Exit Set jso = Nothing Set pdfDoc = Nothing Set AcroExchAVDoc = Nothing Set AcroExchApp = Nothing If fso.FileExists(FilePathJPG) Then ConvertPDFToJPG = FilePathJPG Else ConvertPDFToJPG = "" End If Set fso = Nothing Exit Function ErrorHandler: ConvertPDFToJPG = "" Resume CleanUp End Function ' ========================================================== ' 辅助函数：清理文件名和工作表名中的非法字符 ' ========================================================== Function CleanFileName(ByVal s As String) As String Dim badChars As Variant, i As Long badChars = Array("\", "/", ":", "*", "?", """", "<", ">", "|") For i = LBound(badChars) To UBound(badChars) s = Replace(s, badChars(i), "") Next i CleanFileName = Trim(s) End Function Function CleanSheetName(ByVal s As String) As String Dim badChars As Variant, i As Long badChars = Array("\", "/", "?", "*", "[", "]", ":") For i = LBound(badChars) To UBound(badChars) s = Replace(s, badChars(i), "") Next i CleanSheetName = Trim(s) End Function
